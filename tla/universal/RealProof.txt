
example (c : RealNumbers.Constants) (b : Behavior) (n : Nat) :
    truth (RealNumbers.Same c b n) := by
  simp only [RealNumbers.Same, truth_boolean, RealNumbers.Sum, RealNumbers.Half,
    realBinary_real, integer]
  congr 1
  norm_num

example (c : RealNumbers.Constants) (b : Behavior) (n : Nat) :
    truth (RealNumbers.Bound c b n) := by
  exact realLess_real_infinity ((100 : Int) : Real)

example (c : RealNumbers.Constants) (b : Behavior) (n : Nat) :
    truth (RealNumbers.Lower c b n) := by
  simp only [RealNumbers.Lower, realNegate_infinity]
  exact realLess_minusInfinity_real ((0 : Int) : Real)

example (c : RealNumbers.Constants) (b : Behavior) (n : Nat) :
    truth (RealNumbers.DoubleNegative c b n) := by
  simp [RealNumbers.DoubleNegative, truth_boolean, realNegate_infinity, realNegate_minusInfinity]
