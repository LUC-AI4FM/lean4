/-! The translated source supports an invariant proved for every behavior. -/
namespace TLAImported.Counter

theorem safety (c : Constants) (b : TLA.Behavior State) (h : Spec c b 0) :
    ∀ n, Inv c b n := by
  intro n
  change 0 ≤ (b n).x
  change (b 0).x = 0 ∧ _ at h
  induction n with
  | zero => omega
  | succ n ih =>
    rcases h.2 n (Nat.zero_le n) with step | stutter
    · omega
    · omega
end TLAImported.Counter
